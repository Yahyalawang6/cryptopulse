        function initAudioSynth() {
            try {
                synth = new Tone.Synth().toDestination();
            } catch(e) {
                console.log("Audio Synth Init Error: ", e);
            }
        }

        function playAlertSound(isWin = true) {
            if (!soundEnabled) return;
            try {
                if (Tone.context.state !== 'running') {
                    Tone.start();
                }
                const now = Tone.now();
                if (isWin) {
                    synth.triggerAttackRelease("C5", "8n", now);
                    synth.triggerAttackRelease("G5", "8n", now + 0.15);
                } else {
                    synth.triggerAttackRelease("E3", "8n", now);
                }
            } catch(e) {
                console.log("Audio playback deferred", e);
            }
        }

        function toggleSound() {
            soundEnabled = !soundEnabled;
            const soundBtn = document.getElementById('sound-btn');
            if (soundEnabled) {
                soundBtn.innerHTML = '<i class="fa-solid fa-volume-high text-sm"></i>';
                soundBtn.classList.remove('text-zinc-500');
                soundBtn.classList.add('text-gold');
            } else {
                soundBtn.innerHTML = '<i class="fa-solid fa-volume-xmark text-sm"></i>';
                soundBtn.classList.remove('text-gold');
                soundBtn.classList.add('text-zinc-500');
            }
        }

        function startBinaryTimer(seconds) {
            clearInterval(countdownInterval);
            let timer = seconds;
            const timerDisplay = document.getElementById('binary-countdown');

            countdownInterval = setInterval(() => {
                const secs = timer < 10 ? '0' + timer : timer;
                timerDisplay.innerText = `00:${secs}s`;
                if (--timer < 0) {
                    clearInterval(countdownInterval);
                    timerDisplay.innerText = "EXPIRED";
                    playAlertSound(true);
                }
            }, 1000);
        }

        function initCanvasChart() {
            const canvas = document.getElementById('tradingChart');
            if (!canvas) return;
            const ctx = canvas.getContext('2d');

            function resizeCanvas() {
                canvas.width = canvas.parentElement.clientWidth;
                canvas.height = canvas.parentElement.clientHeight;
                drawChart();
            }

            window.addEventListener('resize', resizeCanvas);
            setTimeout(resizeCanvas, 100);

            function drawChart() {
                const w = canvas.width;
                const h = canvas.height;
                ctx.clearRect(0, 0, w, h);

                // Grid lines
                ctx.strokeStyle = '#18181b';
                ctx.lineWidth = 1;
                for (let x = 0; x < w; x += 40) {
                    ctx.beginPath();
                    ctx.moveTo(x, 0);
                    ctx.lineTo(x, h);
                    ctx.stroke();
                }
                for (let y = 0; y < h; y += 40) {
                    ctx.beginPath();
                    ctx.moveTo(0, y);
                    ctx.lineTo(w, y);
                    ctx.stroke();
                }

                // Generate Dummy Candles
                const candleCount = 28;
                const candleWidth = (w - 60) / candleCount;
                let lastClose = h / 2;

                for (let i = 0; i < candleCount; i++) {
                    const isGreen = Math.random() > 0.45;
                    const candleH = 15 + Math.random() * 45;
                    const open = lastClose;
                    const close = isGreen ? open - candleH : open + candleH;
                    const high = Math.min(open, close) - Math.random() * 15;
                    const low = Math.max(open, close) + Math.random() * 15;
                    const x = 20 + i * candleWidth;

                    // Wick
                    ctx.strokeStyle = isGreen ? '#00E676' : '#FF5252';
                    ctx.beginPath();
                    ctx.moveTo(x + candleWidth / 2, high);
                    ctx.lineTo(x + candleWidth / 2, low);
                    ctx.stroke();

                    // Body
                    ctx.fillStyle = isGreen ? '#00E676' : '#FF5252';
                    ctx.fillRect(x + 2, Math.min(open, close), candleWidth - 4, Math.abs(close - open));

                    lastClose = close;

                    // Draw BUY/SELL overlay on last candle
                    if (i === candleCount - 2) {
                        ctx.fillStyle = '#FFD700';
                        ctx.font = 'bold 10px monospace';
                        ctx.fillText('SIGNAL BUY ▲', x - 20, high - 10);
                    }
                }
            }
        }
